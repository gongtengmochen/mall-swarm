# Docker 环境常用指令
# 进入目录：E:\学校\上课\软件工程期末\mall-swarm\document\docker

# 1. 一键启动所有环境（后台运行）
docker-compose -f docker-compose-env.yml up -d

# 2. 一键暂停所有服务（保留容器，下次快速启动）
docker-compose -f docker-compose-env.yml stop

# 3. 一键关闭并删除容器（数据不丢）
docker-compose -f docker-compose-env.yml down

# 4. 查看服务运行状态
docker-compose -f docker-compose-env.yml ps

# 5. 重启所有服务
docker-compose -f docker-compose-env.yml restart

# 6. 单独启动某个服务（示例：mysql）
docker-compose -f docker-compose-env.yml start mysql

# 7. 单独停止某个服务（示例：mysql）
docker-compose -f docker-compose-env.yml stop mysql

# 8. 单独重启某个服务（示例：mysql）
docker-compose -f docker-compose-env.yml restart mysql
